

<!DOCTYPE html>
<html>
  <head>
    <title>Add Users</title>
    <script src="https://unpkg.com/vue@3.4.18/dist/vue.global.js"></script>
  </head>
  <body>
    <div id="app">
      <user-list></user-list>
    </div>

    <script>
      // Define the UserList component
      const UserList = {
        data() {
          return {
            firstName: '',
            lastName: '',
            users: []  // Store users in an array
          };
        },
        methods: {
          addUser() {
            // Only add user if both fields have values
            if (this.firstName && this.lastName) {
              this.users.push({
                firstName: this.firstName,
                lastName: this.lastName
              });
              this.firstName = '';  // Clear the first name input
              this.lastName = '';    // Clear the last name input
            }
          }
        },
        template: `
          <div>
            First name: <input type="text" v-model="firstName"/> 
            Last name: <input type="text" v-model="lastName"/>
            <button @click="addUser">Add user</button>

            <table>
              <caption>Users</caption>
              <thead>
                <tr>
                  <th>Index</th>
                  <th>First name</th>
                  <th>Last name</th>
                </tr>
              </thead>
              <tbody>
                <tr v-for="(user, index) in users" :key="index">
                  <td>{{ index + 1 }}</td>
                  <td>{{ user.firstName }}</td>
                  <td>{{ user.lastName }}</td>
                </tr>
              </tbody>
            </table>
          </div>
        `
      };

      // Create the Vue app and register the component
      const app = Vue.createApp({});
      app.component("user-list", UserList);
      app.mount("#app");
    </script>
  </body>
</html>
