class Solution {
    public int[] solution(int[] A, int K) {
        // Example: rotate array K times
        int N = A.Length;
        int[] result = new int[N];
        for (int i = 0; i < N; i++) {
            result[(i + K) % N] = A[i];
        }
        return result;
    }
}

int BinarySearch(int[] A, int target) {
    int l = 0, r = A.Length - 1;
    while (l <= r) {
        int m = (l + r) / 2;
        if (A[m] == target) return m;
        if (A[m] < target) l = m + 1;
        else r = m - 1;
    }
    return -1;
}

int Fib(int n) {
    if (n <= 1) return n;
    int a = 0, b = 1;
    for (int i = 2; i <= n; i++) {
        int temp = a + b;
        a = b;
        b = temp;
    }
    return b;
}

int[] PrefixSum(int[] A) {
    int[] P = new int[A.Length + 1];
    for (int i = 0; i < A.Length; i++)
        P[i+1] = P[i] + A[i];
    return P;
}

14. USEFUL CODE PATTERNS

    if (!freq.ContainsKey(num)) freq[num] = 0;

TWO POINTERS:

while (left < right) {
    // logic
    left++;

SLIDING WINDOW:
int sum = 0, k = 3;
for (int i = 0; i < A.Length; i++) {
    sum += A[i];
    if (i >= k) sum -= A[i - k];
}
