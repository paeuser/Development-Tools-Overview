using System;
using System.Collections.Generic;
using System.Linq;

// Define the Claim class
public class Claim
{
    public int Id { get; set; }
    public double Amount { get; set; }
    public string Status { get; set; } // "Open" or "Closed"
    public string Insurer { get; set; } // e.g. "AllState", "Geico"
}

class Program
{
    static void Main()
    {
        // Sample list of claims
        List<Claim> claims = new List<Claim>
        {
            new Claim { Id = 1, Amount = 1200, Status = "Open", Insurer = "AllState" },
            new Claim { Id = 2, Amount = 800, Status = "Closed", Insurer = "Geico" },
            new Claim { Id = 3, Amount = 1500, Status = "Open", Insurer = "AllState" },
            new Claim { Id = 4, Amount = 600, Status = "Closed", Insurer = "Geico" },
            new Claim { Id = 5, Amount = 2000, Status = "Open", Insurer = "StateFarm" },
        };

        // 🔍 LINQ: Filter open claims and group by insurer with total amount
        var openClaimTotals = claims
            .Where(c => c.Status == "Open")
            .GroupBy(c => c.Insurer)
            .Select(group => new
            {
                Insurer = group.Key,
                TotalAmount = group.Sum(c => c.Amount)
            });

        // Display results
        foreach (var item in openClaimTotals)
        {
            Console.WriteLine($"Insurer: {item.Insurer}, Total Open Claim Amount: {item.TotalAmount}");
        }
    }
}
