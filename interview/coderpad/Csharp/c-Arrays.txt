using System;

// To execute C#, please define "static void Main" on a class
// named Solution.

class Solution
{
    
     public string ReverseString(string input)
        {
            char[] charArray = input.ToCharArray(); // Convert string to character array
            Array.Reverse(charArray);               // Reverse the array
            return new string(charArray);           // Convert back to string
        }


    static void Main(string[] args)
    {
       Claim[] claims = new Claim[] {
            new Claim { Id = 1, Amount = 1000, Status = "Open" },
            new Claim { Id = 2, Amount = 500, Status = "Closed" },
            new Claim { Id = 3, Amount = 1500, Status = "Open" }
        };

        foreach (var claim in claims) {
            if (claim.Amount > 1000) {
                Console.WriteLine($"Claim {claim.Id} is greater than 1000: {claim.Amount}");
            }
                
          var solution = new Solution();
          Console.WriteLine(solution.ReverseString("eric").ToString());

        }
    }
}
