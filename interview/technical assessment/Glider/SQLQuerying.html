UPDATE Statement

UPDATE employees 
SET salary = 92000 
WHERE employee_id = 105;

DELETE Statement

DELETE FROM employees 
WHERE employee_id = 104;

🗑️ Deletes David Brown from the employees table.

RANK() with PARTITION BY

SELECT 
  employee_id,
  department_id,
  salary,
  RANK() OVER (PARTITION BY department_id ORDER BY salary DESC) AS salary_rank
FROM employees;

Moving Average with ROWS BETWEEN

SELECT 
  employee_id,
  salary,
  AVG(salary) OVER (
    ORDER BY salary
    ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
  ) AS moving_avg_salary
FROM employees;

Running Average with RANGE

SELECT 
  employee_id,
  salary,
  AVG(salary) OVER (
    ORDER BY salary
    RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ) AS running_avg_salary
FROM employees;
