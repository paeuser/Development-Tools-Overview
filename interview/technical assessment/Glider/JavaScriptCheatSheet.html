Function Practice: Logic Problems

// Sum all numbers in an array function sumArray(arr) { return arr.reduce((total, num) => total + num, 0); } // Reverse a string function reverseString(str) { return str.split('').reverse().join(''); } // Filter array to only even numbers function filterEvens(arr) { return arr.filter(num => num % 2 === 0); } // Check if a string is a palindrome (reads same forwards & backwards) function isPalindrome(str) { const cleaned = str.toLowerCase().replace(/[^a-z0-9]/g, ''); return cleaned === cleaned.split('').reverse().join(''); } // Find max number in an array function findMax(arr) { return Math.max(...arr); } // Example usage console.log(sumArray([1,2,3,4])); // 10 console.log(reverseString('hello')); // 'olleh' console.log(filterEvens([1,2,3,4,5,6])); // [2,4,6] console.log(isPalindrome('Racecar')); // true console.log(findMax([10,5,8,22,3])); // 22
