	<h1>JavaScript DOM Manipulation Tutorial</h1>
	<p>This tutorial demonstrates native DOM methods: <code>querySelector</code>, <code>appendChild</code>, <code>addEventListener</code>, and element manipulation.</p>

	<input type="text" id="inputText" placeholder="Type something...">
	<button id="addBtn">Add Text</button>
	<button id="clearBtn">Clear</button>
	<div id="output"></div>

	<script>
		// Select elements using querySelector
		const addBtn = document.querySelector('#addBtn');
		const clearBtn = document.querySelector('#clearBtn');
		const inputText = document.querySelector('#inputText');
		const output = document.querySelector('#output');

		// Add event listeners for button clicks
		addBtn.addEventListener('click', function() {
			// Check if the input field has a value
			if (inputText.value.trim() !== '') {
				// Create a new paragraph element
				const p = document.createElement('p');
				// Set the text content of the new paragraph to the input value
				p.textContent = inputText.value;
				// Append the new paragraph to the output div, making it visible on the page
				output.appendChild(p);
				// Clear the input field for the next entry
				inputText.value = '';
			}
		});

		clearBtn.addEventListener('click', function() {
			// Set the innerHTML of the output div to an empty string to remove all its children
			output.innerHTML = '';
		});

		// --- Another sample using appendChild ---
		// Create a new div and append it to the body when the page loads
		const sampleDiv = document.createElement('div');
		sampleDiv.textContent = 'This div was added dynamically using appendChild!';
		sampleDiv.style.background = '#d0f0d0';
		sampleDiv.style.padding = '0.75em';
		sampleDiv.style.borderRadius = '8px';
		sampleDiv.style.marginTop = '1em';
		sampleDiv.style.border = '1px solid #a0d0a0';
		document.body.appendChild(sampleDiv);
	</script>
</body>
</html>
